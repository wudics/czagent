### 示例 1 · 最小脚本（无模型依赖）

```ts
export default async function main(ctx) {
  ctx.log('你好，脚本编排！');
  ctx.log('会话：', ctx.session.id);
  ctx.log('工作目录：', ctx.session.cwd);
  return { ok: true, cwd: ctx.session.cwd };
}
```

预期：日志流 3 行 + `✅ 返回值 {"ok":true,...}`。

### 示例 2 · 统计报告（glob + bash + agent.run + write，需 API Key）

```ts
export default async function main(ctx) {
  const files = await ctx.tools.glob({ pattern: 'src/**/*.ts' });
  ctx.log('发现 ' + files.length + ' 个源文件');

  let log = '';
  try {
    log = await ctx.tools.bash({ command: 'git log --oneline -5' });
  } catch (e) {
    log = '（非 git 目录：' + String(e) + '）';
  }

  const summary = await ctx.agent.run('总结这个项目的用途与主要模块，50 字以内');

  if (files.length > 0) {
    await ctx.tools.write({
      file: 'report.md',
      content: '# 项目报告\n\n- 源文件数：' + files.length + '\n\n## 摘要\n\n' + summary,
    });
    ctx.log('已写入 report.md');
  }
  return { files: files.length, summary };
}
```

预期：glob 工具卡片 → bash 卡片（首次弹权限卡）→ `[agent]` 流式输出 → report.md 写入 → 返回值。

### 示例 3 · 联网调研（websearch + webfetch，无 API Key）

```ts
export default async function main(ctx) {
  const hits = await ctx.tools.websearch({ query: 'electron 32 新特性' });
  ctx.log(hits);
  const m = /https?:\/\/\S+/.exec(hits);
  if (!m) return '搜索无结果';
  const page = await ctx.tools.webfetch({ url: m[0] });
  return page.slice(0, 1000);
}
```

预期：搜索结果列表入日志 → webfetch 抓详情（超长自动落盘+预览）→ 返回前 1000 字。

### 示例 5 · 中止感知循环（无 API Key，运行中点 ■ 测试停止）

```ts
export default async function main(ctx) {
  for (let i = 1; i <= 30; i++) {
    if (ctx.signal.aborted) {
      ctx.log('第 ' + i + ' 步前检测到停止信号');
      return '已中止（完成 ' + (i - 1) + ' 步）';
    }
    await new Promise((r) => setTimeout(r, 100));
    ctx.log('步骤 ' + i + ' 完成');
  }
  return '全部完成';
}
```

预期：日志逐行流出；点停止后下一轮检测到信号 → 优雅返回 `已中止（完成 N 步）`。

### 示例 6 · 权限路径（cwd 内放行 / cwd 外弹卡 / 显式 ask，无 API Key）

```ts
export default async function main(ctx) {
  await ctx.tools.write({ file: '脚本输出.txt', content: '无需确认' });
  ctx.log('cwd 内写入完成（直接放行）');

  try {
    await ctx.tools.write({ file: '../脚本外写.txt', content: '需要确认' });
    ctx.log('cwd 外写入完成（已允许）');
  } catch (e) {
    ctx.log('cwd 外写入被拒绝：', String(e));
  }

  const go = await ctx.ask({ tool: 'bash', args: { command: 'echo 继续' } });
  if (go === 'allow' || go === 'allowAlways') {
    ctx.log(await ctx.tools.bash({ command: 'echo 继续' }));
  }
  return 'done';
}
```

预期：cwd 内直写；cwd 外写弹权限卡（拒绝 → catch 捕获继续）；显式 ask 弹卡。

## 实施后发现并修复（示例脚本冒烟暴露）

- **外部停止时脚本无法优雅退出**：`runScript` 的 race loser 在外部 abort 时**立即 reject**（且超时文案为创建时求值的误导信息），抢在脚本感知 `ctx.signal` 优雅返回之前。修复：外部中止仅联动 `ctx.signal`（脚本立即感知），**5 秒宽限期**内未返回才强制 reject；超时文案改为触发时求值。示例 5/集成 Stop 用例据此验证优雅路径。
